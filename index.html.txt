        // --- Lógica del Calendario y Dashboard ---

        let currentActiveSection = 'dashboard';
        let currentMonth = new Date().getMonth();
        let currentYear = new Date().getFullYear();
        let currentFilter = 'all';

        // Helper para obtener el nombre del mes
        const MONTH_NAMES = [
            'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
            'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'

        // --- Funciones de Utilidad ---

        function formatDate(dateString) {
            const date = new Date(dateString + 'T00:00:00'); // Añadir T00:00:00 para evitar problemas de zona horaria
            return date.toLocaleDateString('es-ES', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });
        }

        function calculateDaysUntil(dateString, timeString) {
            const matchDateTime = new Date(`${dateString}T${timeString}:00`);
            const now = new Date();
            const diffTime = matchDateTime.getTime() - now.getTime();
            
            if (diffTime < 0) return 'JUGADO';
            
            const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
            return diffDays === 0 ? 'HOY' : `${diffDays} DÍA${diffDays !== 1 ? 'S' : ''}`;
        }
        
        function getNextMatch() {
            const now = new Date();
            // Filtrar partidos futuros y ordenarlos
            const upcomingMatches = MATCHES_DATA.filter(match => {
                const matchDateTime = new Date(`${match.date}T${match.time}:00`);
                return matchDateTime > now;
            }).sort((a, b) => {
                const dateA = new Date(`${a.date}T${a.time}:00`);
                const dateB = new Date(`${b.date}T${b.time}:00`);
                return dateA - dateB;
            });
            
            return upcomingMatches[0] || null;
        }

        // --- 1A. Dashboard (Inicio) ---

        function renderDashboard() {
            const nextMatch = getNextMatch();
            const detailsElement = document.getElementById('next-match-details');
            const countdownElement = document.getElementById('countdown-timer');
            const listElement = document.getElementById('upcoming-matches-list');

            if (nextMatch) {
                const daysUntil = calculateDaysUntil(nextMatch.date, nextMatch.time);
                detailsElement.innerHTML = `
                    **${nextMatch.modality}** vs **${nextMatch.rival}**<br>
                    ${formatDate(nextMatch.date)} a las ${nextMatch.time} en ${nextMatch.location}
                `;
                countdownElement.textContent = daysUntil;
            } else {
                detailsElement.textContent = 'No hay partidos programados próximamente.';
                countdownElement.textContent = 'FIN';
            }

            // Listar próximos 5 partidos
            listElement.innerHTML = '';
            MATCHES_DATA.filter(match => {
                const matchDateTime = new Date(`${match.date}T${match.time}:00`);
                return matchDateTime > new Date();
            })
            .sort((a, b) => new Date(`${a.date}T${a.time}:00`) - new Date(`${b.date}T${b.time}:00`))
            .slice(0, 5)
            .forEach(match => {
                const li = document.createElement('li');
                li.innerHTML = `
                    <i class="fas fa-arrow-circle-right"></i>
                    **${match.modality}** vs ${match.rival} | 
                    ${formatDate(match.date).split(', ')[0]}, ${match.time}
                    (<span style="font-weight: bold; color: var(--color-primary);">${calculateDaysUntil(match.date, match.time)}</span>)
                    <button class="btn btn-secondary" style="float: right; padding: 3px 8px;" onclick="showMatchDetail('${match.date}', '${match.time}')">Detalles</button>
                `;
                listElement.appendChild(li);
            });
        }

        // --- 1B. Calendario Interactivo ---

        function renderCalendar() {
            const grid = document.getElementById('calendar-grid');
            const monthYearTitle = document.getElementById('current-month-year');
            const firstDayOfMonth = new Date(currentYear, currentMonth, 1);
            const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
            //getDay() devuelve 0 (Dom) a 6 (Sab). Necesitamos 0 (Lun) a 6 (Dom).
            let startDay = firstDayOfMonth.getDay();
            if (startDay === 0) startDay = 7; // Mueve Domingo (0) al final (7)
            startDay = startDay - 1; // Ahora 0 es Lunes, 6 es Domingo.

            monthYearTitle.textContent = `${MONTH_NAMES[currentMonth]} ${currentYear}`;
            grid.innerHTML = `
                <div class="calendar-day-label">Lun</div>
                <div class="calendar-day-label">Mar</div>
                <div class="calendar-day-label">Mié</div>
                <div class="calendar-day-label">Jue</div>
                <div class="calendar-day-label">Vie</div>
                <div class="calendar-day-label">Sáb</div>
                <div class="calendar-day-label">Dom</div>
            `;

            // Días del mes anterior (padding)
            for (let i = 0; i < startDay; i++) {
                const dayDiv = document.createElement('div');
                dayDiv.classList.add('calendar-day', 'not-current-month');
                grid.appendChild(dayDiv);
            }

            // Días del mes actual
            for (let day = 1; day <= daysInMonth; day++) {
                const dateString = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
                
                // Filtrar partidos para el día
                const matchesOfTheDay = MATCHES_DATA.filter(match => 
                    match.date === dateString && 
                    (currentFilter === 'all' || match.modality === currentFilter)
                );

                const dayDiv = document.createElement('div');
                dayDiv.classList.add('calendar-day');
                dayDiv.innerHTML = `<div class="day-number">${day}</div>`;

                if (matchesOfTheDay.length > 0) {
                    dayDiv.classList.add('has-match');
                    matchesOfTheDay.forEach(match => {
                        const indicator = document.createElement('span');
                        indicator.classList.add('match-indicator');
                        indicator.textContent = `${match.modality.split(' ')[0][0]}${match.modality.split(' ')[1][0]} vs ${match.rival}`;
                        // Añadir evento al indicador para mostrar detalles
                        indicator.onclick = (e) => {
                            e.stopPropagation();
                            showMatchDetail(match.date, match.time);
                        };
                        dayDiv.appendChild(indicator);
                    });
                }
                
                grid.appendChild(dayDiv);
            }
        }

        function changeMonth(delta) {
            currentMonth += delta;
            if (currentMonth < 0) {
                currentMonth = 11;
                currentYear--;
            } else if (currentMonth > 11) {
                currentMonth = 0;
                currentYear++;
            }
            renderCalendar();
        }

        function setFilter(modality) {
            currentFilter = modality;
            document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelector(`.filter-btn[data-filter="${modality}"]`).classList.add('active');
            renderCalendar();
        }

        // --- 1C. Ficha Detallada de Partido (Modal) ---

        function showMatchDetail(date, time) {
            const match = MATCHES_DATA.find(m => m.date === date && m.time === time);
            if (!match) return;

            const modal = document.getElementById('match-detail-modal');
            const ownTeam = TEAMS[match.modality];

            document.getElementById('modal-match-modality').textContent = match.modality;
            document.getElementById('modal-own-logo').src = ownTeam.escudo;
            document.getElementById('modal-rival-name').textContent = match.rival;
            // No tenemos escudos de rivales, usamos un placeholder.
            document.getElementById('modal-rival-logo').src = 'img/rival_default.png'; 
            document.getElementById('modal-match-date').textContent = formatDate(match.date);
            document.getElementById('modal-match-time').textContent = match.time;
            document.getElementById('modal-match-location').textContent = match.location;
            document.getElementById('modal-match-result').textContent = match.result || 'Pendiente';
            
            // Enlace de Google Calendar
            const gcalLink = createGoogleCalendarLink(match);
            document.getElementById('export-gcal').onclick = () => window.open(gcalLink, '_blank');

            modal.style.display = 'flex';
        }

        function closeMatchDetail() {
            document.getElementById('match-detail-modal').style.display = 'none';
        }

        function createGoogleCalendarLink(match) {
            const dateStr = match.date.replace(/-/g, '');
            const timeStr = match.time.replace(':', '') + '00';
            const start = `${dateStr}T${timeStr}`;

            // Asumimos que la duración es de 1 hora y 30 minutos (1h30m)
            const endHour = parseInt(match.time.split(':')[0]) + 1;
            const endMinute = parseInt(match.time.split(':')[1]) + 30;
            const endTime = `${String(endHour).padStart(2, '0')}${String(endMinute).padStart(2, '0')}00`;
            const end = `${dateStr}T${endTime}`;

            const details = `Partido de ${match.modality} del Trofeo Rector. EUE Huesca vs ${match.rival}.`;
            const location = match.location;
            const text = `Partido: EUE Huesca vs ${match.rival} (${match.modality})`;

            const baseUrl = 'https://www.google.com/calendar/render?action=TEMPLATE';
            const params = new URLSearchParams({
                text: text,
                dates: `${start}/${end}`,
                details: details,
                location: location,
                sf: 'true',
                output: 'xml'
            });

            return `${baseUrl}&${params.toString()}`;
        }

        function navigateTo(sectionId) {
            currentActiveSection = sectionId;
            document.querySelectorAll('.content-section').forEach(section => {
                section.style.display = 'none';
            });
            document.getElementById(sectionId).style.display = 'block';

            document.querySelectorAll('.nav-link').forEach(link => {
                link.classList.remove('active');
            });

        // --- Identidad Visual: Dark Mode ---

        function toggleDarkMode() {
            document.body.classList.toggle('dark-mode');
            const isDarkMode = document.body.classList.contains('dark-mode');
            localStorage.setItem('darkMode', isDarkMode);
            
            // Icono del toggle
            const toggleIcon = document.querySelector('.dark-mode-toggle');
            if (isDarkMode) {
                toggleIcon.classList.replace('fa-moon', 'fa-sun');
            } else {
                toggleIcon.classList.replace('fa-sun', 'fa-moon');
            }
        }

        function checkDarkModePreference() {
            const savedMode = localStorage.getItem('darkMode');
            if (savedMode === 'true') {
                toggleDarkMode(); // Llama a toggle para aplicar el modo y actualizar el icono
            } else if (savedMode === null && window.matchMedia('(prefers-color-scheme: dark)').matches) {
                 toggleDarkMode(); // Aplica por defecto si el sistema es oscuro
            }
        }
